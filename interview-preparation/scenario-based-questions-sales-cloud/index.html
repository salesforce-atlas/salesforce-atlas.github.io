    trigger LeadReferralAssignment on Lead (before insert, before update) {
    Set<Id> referringRepIds = new Set<Id>();

    for (Lead lead : Trigger.new) {
        if (lead.Lead_Source_Detail__c == 'Referral' && lead.Referring_Rep__c != null) {
            referringRepIds.add(lead.Referring_Rep__c);
        }
    }

    // Query manager information
    Map<Id, User> userManagerMap = new Map<Id, User>();
    for (User u : [SELECT Id, ManagerId FROM User WHERE Id IN :referringRepIds]) {
        userManagerMap.put(u.Id, u);
    }

    // Assign to manager
    for (Lead lead : Trigger.new) {
        if (lead.Lead_Source_Detail__c == 'Referral' && lead.Referring_Rep__c != null) {
            User referringUser = userManagerMap.get(lead.Referring_Rep__c);
            if (referringUser != null && referringUser.ManagerId != null) {
                lead.OwnerId = referringUser.ManagerId;
            }
        }
    }
}

    public class LeadScoringService {

    // Scoring values
    private static final Integer EMAIL_OPEN_POINTS = 5;
    private static final Integer WEBSITE_VISIT_POINTS = 10;
    private static final Integer DEMO_REQUEST_POINTS = 50;
    private static final Integer PRICING_PAGE_POINTS = 20;
    private static final Integer MQL_THRESHOLD = 100;

    // Calculate lead score based on activities
    public static void calculateLeadScore(List<Id> leadIds) {
        Map<Id, Integer> leadScores = new Map<Id, Integer>();

        // Query all relevant activities
        List<Task> activities = [
            SELECT WhoId, Subject, ActivityDate, Type
            FROM Task
            WHERE WhoId IN :leadIds
            AND ActivityDate = LAST_90_DAYS
        ];

        // Calculate scores
        for (Task activity : activities) {
            Integer currentScore = leadScores.containsKey(activity.WhoId) 
                ? leadScores.get(activity.WhoId) : 0;

            if (activity.Subject != null) {
                if (activity.Subject.contains('Email Opened')) {
                    currentScore += EMAIL_OPEN_POINTS;
                } else if (activity.Subject.contains('Website Visit')) {
                    currentScore += WEBSITE_VISIT_POINTS;
                } else if (activity.Subject.contains('Demo Request')) {
                    currentScore += DEMO_REQUEST_POINTS;
                } else if (activity.Subject.contains('Pricing Page')) {
                    currentScore += PRICING_PAGE_POINTS;
                }
            }

            leadScores.put(activity.WhoId, currentScore);
        }

        // Update leads
        updateLeadScores(leadScores);
    }

    // Update lead records with new scores
    private static void updateLeadScores(Map<Id, Integer> leadScores) {
        List<Lead> leadsToUpdate = new List<Lead>();

        for (Id leadId : leadScores.keySet()) {
            Lead l = new Lead(
                Id = leadId,
                Lead_Score__c = leadScores.get(leadId),
                Last_Score_Update__c = System.now()
            );
            leadsToUpdate.add(l);
        }

        if (!leadsToUpdate.isEmpty()) {
            update leadsToUpdate;
        }
    }

    // Check for MQL qualification
    public static void checkMQLQualification(List<Lead> leads) {
        List<Lead> mqlLeads = new List<Lead>();

        for (Lead l : leads) {
            if (l.Lead_Score__c >= MQL_THRESHOLD && 
                (l.Lead_Grade__c == 'A' || l.Lead_Grade__c == 'B') &&
                l.MQL_Date__c == null) {

                l.Status = 'Marketing Qualified';
                l.MQL_Date__c = Date.today();
                mqlLeads.add(l);
            }
        }

        if (!mqlLeads.isEmpty()) {
            // Notify sales team
            notifySalesTeam(mqlLeads);
        }
    }

    // Send notifications to sales team
    private static void notifySalesTeam(List<Lead> mqlLeads) {
        List<Messaging.SingleEmailMessage> emails = new List<Messaging.SingleEmailMessage>();

        for (Lead l : mqlLeads) {
            Messaging.SingleEmailMessage email = new Messaging.SingleEmailMessage();
            email.setTargetObjectId(l.OwnerId);
            email.setSubject('New MQL: ' + l.Name);
            email.setPlainTextBody('Lead ' + l.Name + ' has qualified as MQL with score: ' + l.Lead_Score__c);
            email.setSaveAsActivity(false);
            emails.add(email);
        }

        if (!emails.isEmpty()) {
            Messaging.sendEmail(emails);
        }
    }
}

    trigger LeadScoringTrigger on Lead (after update) {
    List<Lead> leadsToCheck = new List<Lead>();

    for (Lead l : Trigger.new) {
        Lead oldLead = Trigger.oldMap.get(l.Id);

        // Check if score or grade changed
        if (l.Lead_Score__c != oldLead.Lead_Score__c || 
            l.Lead_Grade__c != oldLead.Lead_Grade__c) {
            leadsToCheck.add(l);
        }
    }

    if (!leadsToCheck.isEmpty()) {
        LeadScoringService.checkMQLQualification(leadsToCheck);
    }
}

    global class LeadScoringBatch implements Database.Batchable<sObject> {

    global Database.QueryLocator start(Database.BatchableContext bc) {
        return Database.getQueryLocator([
            SELECT Id, Lead_Score__c, Lead_Grade__c, MQL_Date__c
            FROM Lead
            WHERE Status != 'Converted'
            AND Status != 'Disqualified'
        ]);
    }

    global void execute(Database.BatchableContext bc, List<Lead> scope) {
        Set<Id> leadIds = new Set<Id>();
        for (Lead l : scope) {
            leadIds.add(l.Id);
        }

        LeadScoringService.calculateLeadScore(new List<Id>(leadIds));
    }

    global void finish(Database.BatchableContext bc) {
        System.debug('Lead scoring batch completed');
    }
}

        List<Quote> quotes = [

            FROM Quote
            WHERE Id IN :quoteIds
        ];

        for (Quote q : quotes) {
            ApprovalRequirement req = new ApprovalRequirement();
            req.quoteId = q.Id;
            req.requiresManagerApproval = (q.TotalPrice >= 50000);
            req.requiresVPApproval = (q.TotalPrice >= 250000);

        }

        return requirementsMap;
    }

    public class TerritoryReassignmentQueueable implements Queueable {

    private List<Account> accounts;

    public TerritoryReassignmentQueueable(List<Account> accounts) {
        this.accounts = accounts;
    }

    public void execute(QueueableContext context) {
        // Get territory assignment rules
        Map<String, Id> stateToTerritoryMap = getStateTerritoryMapping();
        Map<Id, Id> territoryOwnerMap = getTerritoryOwners();

        List<Account> accountsToUpdate = new List<Account>();
        List<ObjectTerritory2Association> newAssociations = new List<ObjectTerritory2Association>();
        List<ObjectTerritory2Association> oldAssociations = new List<ObjectTerritory2Association>();

        // Query existing territory assignments
        Map<Id, ObjectTerritory2Association> existingAssignments = 
            getExistingTerritoryAssignments(accounts);

        for (Account acc : accounts) {
            String state = acc.BillingState;

            if (stateToTerritoryMap.containsKey(state)) {
                Id newTerritoryId = stateToTerritoryMap.get(state);
                Id newOwnerId = territoryOwnerMap.get(newTerritoryId);

                // Remove old territory assignment
                if (existingAssignments.containsKey(acc.Id)) {
                    oldAssociations.add(existingAssignments.get(acc.Id));
                }

                // Create new territory assignment
                ObjectTerritory2Association newAssoc = new ObjectTerritory2Association(
                    ObjectId = acc.Id,
                    Territory2Id = newTerritoryId,
                    AssociationCause = 'Territory2Manual'
                );
                newAssociations.add(newAssoc);

                // Update account owner
                Account accToUpdate = new Account(
                    Id = acc.Id,
                    OwnerId = newOwnerId,
                    Previous_Owner__c = acc.OwnerId,
                    Territory_Change_Date__c = Date.today(),
                    Territory_Change_Reason__c = 'Billing Address Change'
                );
                accountsToUpdate.add(accToUpdate);
            }
        }

        // Perform DML operations
        if (!oldAssociations.isEmpty()) {
            delete oldAssociations;
        }

        if (!newAssociations.isEmpty()) {
            insert newAssociations;
        }

        if (!accountsToUpdate.isEmpty()) {
            update accountsToUpdate;

            // Send notifications
            sendOwnershipChangeNotifications(accountsToUpdate);
        }
    }

    // Get mapping of state to territory
    private Map<String, Id> getStateTerritoryMapping() {
        Map<String, Id> mapping = new Map<String, Id>();

        List<Territory2> territories = [
            SELECT Id, Name, State__c
            FROM Territory2
            WHERE State__c != null
        ];

        for (Territory2 terr : territories) {
            mapping.put(terr.State__c, terr.Id);
        }

        return mapping;
    }

    // Get territory owners
    private Map<Id, Id> getTerritoryOwners() {
        Map<Id, Id> ownerMap = new Map<Id, Id>();

        List<UserTerritory2Association> userTerritories = [
            SELECT Territory2Id, UserId, RoleInTerritory2
            FROM UserTerritory2Association
            WHERE RoleInTerritory2 = 'Account Owner'
            AND IsActive = true
        ];

        for (UserTerritory2Association uta : userTerritories) {
            ownerMap.put(uta.Territory2Id, uta.UserId);
        }

        return ownerMap;
    }

    // Get existing territory assignments
    private Map<Id, ObjectTerritory2Association> getExistingTerritoryAssignments(
        List<Account> accounts) {

        Map<Id, ObjectTerritory2Association> assignmentMap = 
            new Map<Id, ObjectTerritory2Association>();

        Set<Id> accountIds = new Set<Id>();
        for (Account acc : accounts) {
            accountIds.add(acc.Id);
        }

        List<ObjectTerritory2Association> associations = [
            SELECT Id, ObjectId, Territory2Id
            FROM ObjectTerritory2Association
            WHERE ObjectId IN :accountIds
        ];

        for (ObjectTerritory2Association assoc : associations) {
            assignmentMap.put(assoc.ObjectId, assoc);
        }

        return assignmentMap;
    }

    // Send notifications about ownership change
    private void sendOwnershipChangeNotifications(List<Account> accounts) {
        List<Messaging.SingleEmailMessage> emails = new List<Messaging.SingleEmailMessage>();

        for (Account acc : accounts) {
            // Email to new owner
            Messaging.SingleEmailMessage newOwnerEmail = new Messaging.SingleEmailMessage();
            newOwnerEmail.setTargetObjectId(acc.OwnerId);
            newOwnerEmail.setSaveAsActivity(false);

            newOwnerEmail.setPlainTextBody(
                'Account ' + acc.Name + ' has been assigned to you due to territory change.\n\n' +
                'Previous Owner: ' + acc.Previous_Owner__c + '\n' +
                'New Territory: ' + acc.BillingState

            // Email to previous owner

                Messaging.SingleEmailMessage prevOwnerEmail = new Messaging.SingleEmailMessage();

            }
        }

        if (!emails.isEmpty() && !Test.isRunningTest()) {
            Messaging.sendEmail(emails);

    }

    public static Decimal calculateBundleTotal(Id bundleProductId, Id pricebookId) {
        Decimal total = 0;

        List<Product_Bundle_Item__c> bundleItems = [
            SELECT Component_Product__c, Quantity__c, Discount_Percentage__c
            FROM Product_Bundle_Item__c
            WHERE Bundle_Product__c = :bundleProductId
        ];

        Set<Id> componentIds = new Set<Id>();
        for (Product_Bundle_Item__c item : bundleItems) {
            componentIds.add(item.Component_Product__c);
        }

        Map<Id, PricebookEntry> pbeMap = getPricebookEntries(pricebookId, componentIds);

        for (Product_Bundle_Item__c item : bundleItems) {
            PricebookEntry pbe = pbeMap.get(item.Component_Product__c);
            if (pbe != null) {
                Decimal itemTotal = pbe.UnitPrice * item.Quantity__c;
                Decimal discount = item.Discount_Percentage__c != null ? 
                    item.Discount_Percentage__c / 100 : 0;
                itemTotal = itemTotal * (1 - discount);
                total += itemTotal;
            }
        }

        return total;
    }
}

    public class OpportunityEscalationService {

    // Main method to process stagnant opportunities
    public static void processStagnantOpportunities() {
        List<Opportunity> stagnantOpps = getStagnantOpportunities();

        if (!stagnantOpps.isEmpty()) {
            createEscalationTasks(stagnantOpps);
            sendEscalationEmails(stagnantOpps);
            updateOpportunityEscalationStatus(stagnantOpps);
            postChatterUpdates(stagnantOpps);
            logEscalationHistory(stagnantOpps);
        }
    }

    // Query stagnant opportunities
    private static List<Opportunity> getStagnantOpportunities() {
        Date thirtyDaysAgo = Date.today().addDays(-30);
        Date sevenDaysAgo = Date.today().addDays(-7);

        return [
            SELECT Id, Name, StageName, Amount, Probability, CloseDate,
                   OwnerId, Owner.ManagerId, Owner.Name, Owner.Email,
                   Owner.Manager.Name, Owner.Manager.Email,
                   AccountId, Account.Name,
                   Days_In_Current_Stage__c, Escalation_Count__c,
                   Last_Stage_Change_Date__c
            FROM Opportunity
            WHERE IsClosed = false
            AND Probability < 0.50
            AND Last_Stage_Change_Date__c <= :thirtyDaysAgo
            AND (Last_Escalation_Date__c < :sevenDaysAgo OR Last_Escalation_Date__c = null)
            ORDER BY Days_In_Current_Stage__c DESC
            LIMIT 200
        ];
    }

    // Create tasks for reps
    private static void createEscalationTasks(List<Opportunity> opportunities) {
        List<Task> tasksToCreate = new List<Task>();

        for (Opportunity opp : opportunities) {
            Task escalationTask = new Task(
                Subject = 'URGENT: Follow up on stagnant opportunity - ' + opp.Name,
                Priority = 'High',
                Status = 'Not Started',
                ActivityDate = Date.today().addDays(2),
                OwnerId = opp.OwnerId,
                WhatId = opp.Id,
                Description = buildTaskDescription(opp)
            );
            tasksToCreate.add(escalationTask);
        }

        if (!tasksToCreate.isEmpty()) {
            insert tasksToCreate;
        }
    }

    // Build task description
    private static String buildTaskDescription(Opportunity opp) {
        return 'This opportunity has been in the ' + opp.StageName + 
               ' stage for ' + opp.Days_In_Current_Stage__c + 
               ' days without progress.\n\n' +
               'Account: ' + opp.Account.Name + '\n' +
               'Amount: $' + opp.Amount + '\n' +
               'Probability: ' + opp.Probability + '%\n' +
               'Close Date: ' + opp.CloseDate + '\n\n' +
               'REQUIRED ACTIONS:\n' +
               '1. Contact customer to assess current interest level\n' +
               '2. Identify any blockers preventing progress\n' +
               '3. Update opportunity stage or close date\n' +
               '4. Document next steps\n' +
               '5. If no longer viable, mark as Closed Lost\n\n' +
               'This escalation has been sent to your manager.';
    }

    // Send escalation emails
    private static void sendEscalationEmails(List<Opportunity> opportunities) {
        List<Messaging.SingleEmailMessage> emails = new List<Messaging.SingleEmailMessage>();

        // Get email template
        EmailTemplate template = [
            SELECT Id 
            FROM EmailTemplate 
            WHERE DeveloperName = 'Stagnant_Opportunity_Manager_Alert' 
            LIMIT 1
        ];

        for (Opportunity opp : opportunities) {
            if (opp.Owner.ManagerId != null) {
                // Email to manager
                Messaging.SingleEmailMessage managerEmail = new Messaging.SingleEmailMessage();
                managerEmail.setTargetObjectId(opp.Owner.ManagerId);
                managerEmail.setWhatId(opp.Id);
                managerEmail.setTemplateId(template.Id);
                managerEmail.setSaveAsActivity(false);
                emails.add(managerEmail);

                // Email to rep
                Messaging.SingleEmailMessage repEmail = new Messaging.SingleEmailMessage();
                repEmail.setTargetObjectId(opp.OwnerId);
                repEmail.setWhatId(opp.Id);
                repEmail.setTemplateId(template.Id);
                repEmail.setSaveAsActivity(false);
                emails.add(repEmail);
            }
        }

        if (!emails.isEmpty()) {
            Messaging.sendEmail(emails);
        }
    }

    // Update escalation status
    private static void updateOpportunityEscalationStatus(List<Opportunity> opportunities) {
        List<Opportunity> oppsToUpdate = new List<Opportunity>();

        for (Opportunity opp : opportunities) {
            Opportunity oppUpdate = new Opportunity(
                Id = opp.Id,
                Escalation_Count__c = (opp.Escalation_Count__c != null ? 
                    opp.Escalation_Count__c + 1 : 1),
                Last_Escalation_Date__c = Date.today(),
                Escalation_Status__c = 'Escalated to Manager'
            );
            oppsToUpdate.add(oppUpdate);
        }

        if (!oppsToUpdate.isEmpty()) {
            update oppsToUpdate;
        }
    }

    // Post Chatter updates
    private static void postChatterUpdates(List<Opportunity> opportunities) {
        List<FeedItem> posts = new List<FeedItem>();

        for (Opportunity opp : opportunities) {
            String message = '@[' + opp.Owner.ManagerId + '] - ' +
                           'This opportunity has been stagnant for ' +
                           opp.Days_In_Current_Stage__c + ' days. ' +
                           'Owner: @[' + opp.OwnerId + ']\n\n' +
                           'Amount: $' + opp.Amount + '\n' +
                           'Stage: ' + opp.StageName + '\n' +
                           'Probability: ' + opp.Probability + '%';

            FeedItem post = new FeedItem(
                ParentId = opp.Id,
                Body = message,
                Type = 'TextPost'
            );
            posts.add(post);
        }

        if (!posts.isEmpty()) {
            insert posts;
        }
    }

    // Log escalation history
    private static void logEscalationHistory(List<Opportunity> opportunities) {
        List<Opportunity_Escalation_Log__c> logs = new List<Opportunity_Escalation_Log__c>();

        for (Opportunity opp : opportunities) {
            Opportunity_Escalation_Log__c log = new Opportunity_Escalation_Log__c(
                Opportunity__c = opp.Id,
                Escalation_Date__c = Date.today(),
                Days_Stagnant__c = opp.Days_In_Current_Stage__c,
                Stage_at_Escalation__c = opp.StageName,
                Amount_at_Escalation__c = opp.Amount,
                Escalation_Reason__c = 'No stage change for 30+ days',
                Escalated_To__c = opp.Owner.ManagerId
            );
            logs.add(log);
        }

        if (!logs.isEmpty()) {
            insert logs;
        }
    }
